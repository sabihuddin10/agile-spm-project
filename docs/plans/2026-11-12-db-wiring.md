# DB Wiring Implementation Plan

**Goal:** Persist the in-memory store in the 29 relational tables of `server/schema/001_schema.sql` instead of the gzip JSON blob in `app_state`.
**Architecture:** `persist.js` keeps one transaction per request. It serialises with `pg_advisory_xact_lock`, keeps the store version in `sequences('state_version')`, reloads the store from the tables when another instance saved a newer version, and rewrites the tables from the store when a request changed it. Mappers in `server/src/data/relational/` translate store records ↔ table rows; string ids live in `ext_id`, bigint ids are assigned as array index + 1 so foreign keys resolve in JS.
**Tech Stack:** Node 22, Express 4, `pg` / `@neondatabase/serverless`, `node:test`, PostgreSQL 16 (local) / Neon.

Spec: `docs/specs/2026-11-12-db-wiring-design.md`.

## Known store ↔ schema gaps (resolved in the mappers)

| Store | Schema | Resolution |
|---|---|---|
| `order.subtotal/serviceCharge/tax/total` | not stored (DB_PLAN §3.5) | recomputed with `computeTotals()` on load |
| `order.rates` | `tax_rate`, `service_charge_rate` | split / rebuilt |
| `order.split`, `order.refund` | `bill_splits`/`split_parts`/`split_part_items`, `order_refunds` | child rows |
| `purchaseOrder.number` = `"PO-0001"` | `integer` | stored as `1`, formatted back with `PO-` + 4-digit pad |
| `purchaseOrder.lines[].supplier` | no column | new column, `server/schema/003_store_fields.sql` |
| `notification.readBy[]`, `notification.to` | `notification_reads`, `to_email` | child rows / rename |
| `customer.preferences{dietary,allergies}` | `customer_preferences(kind, value)` | child rows (`allergies` ↔ `allergy`) |
| `counters`, `orderSeq` | `sequences` | rows `id:<prefix>`, `order_seq` |

Order of `readBy` and preference values is not kept by the schema (set semantics); the round-trip test compares them sorted.

/** Order-insensitive fields compared as sorted lists. */
function normalise(snap) {
  const s = structuredClone(snap);
  for (const n of s.notifications) n.readBy.sort();
  for (const c of s.customers) for (const k of Object.keys(c.preferences ?? {})) c.preferences[k].sort();
  return s;
}

test('saving then loading the store gives back the same store', async () => {
  // Arrange
  const before = snapshot();

  // Act
  const client = await pool.connect();
  try {
    await client.query('BEGIN');
    await saveStore(client, before);
    await client.query('COMMIT');
    restore({});
    restore(await loadStore(client));
  } finally { client.release(); }

  // Assert
  assert.deepEqual(normalise(snapshot()), normalise(before));

### Task 3: Row helpers

**Files:**
- Create: `server/src/data/relational/sql.js`

```js
// server/src/data/relational/sql.js
/** Shared helpers for the store ↔ table mappers. */

export const iso = (v) => (v == null ? null : new Date(v).toISOString());
export const hm = (v) => (v == null ? null : String(v).slice(0, 5));
export const num = (v) => (v == null ? null : Number(v));

/** ext id → bigint id, where bigint id = array index + 1. */
export const idsOf = (list) => new Map(list.map((x, i) => [x.id, i + 1]));
/** bigint id → ext id, from loaded rows. */
export const extsOf = (rows) => new Map(rows.map((r) => [r.id, r.ext_id]));
export const ref = (map, key) => (key == null ? null : map.get(key) ?? null);

const names = (defs) => defs.split(',').map((d) => d.trim().split(/\s+/)[0].replaceAll('"', ''));

/** Insert all rows in one statement. `defs` is a column list with types: 'id bigint, name text'. */
export async function insertRows(db, table, defs, rows, { identity = true } = {}) {
  if (!rows.length) return;
  const cols = names(defs).map((n) => `"${n}"`).join(', ');
  const overriding = identity ? 'OVERRIDING SYSTEM VALUE' : '';
  await db.query(
    `INSERT INTO ${table} (${cols}) ${overriding} SELECT ${cols} FROM jsonb_to_recordset($1::jsonb) AS x(${defs})`,
    [JSON.stringify(rows)],
  );
}

/** All rows of a table as JSON objects, in `orderBy` order. */
export async function selectRows(db, table, orderBy = 'id') {
  const { rows } = await db.query(`SELECT coalesce(json_agg(t ORDER BY ${orderBy}), '[]'::json) AS rows FROM ${table} t`);
  return rows[0].rows;
}
```

`json_agg` renders `numeric` as JSON numbers, `date` as `YYYY-MM-DD`, `time` as `HH:MM:SS` (→ `hm`), `timestamptz` with an offset (→ `iso`).

### Task 4: Mappers

**Files:** Create one module per area in `server/src/data/relational/`. Each exports `save(snap, ids)` → `[[table, defs, rows, opts?], …]` in insert order, and `load(rows, exts)` → store fragment. `ids` / `exts` are the shared lookup maps built in `index.js`.

```js
// people.js — users, customers, customer_preferences
import { iso, num, ref } from './sql.js';

export function save(s, ids) {
  return [
    ['users', 'id bigint, ext_id text, name text, email text, phone text, role role, active boolean, must_change_password boolean, password_hash text, token_version integer, hourly_wage numeric, created_at timestamptz',
      s.users.map((u, i) => ({ id: i + 1, ext_id: u.id, name: u.name, email: u.email, phone: u.phone, role: u.role, active: u.active,
        must_change_password: u.mustChangePassword, password_hash: u.passwordHash, token_version: u.tokenVersion, hourly_wage: u.hourlyWage ?? null, created_at: u.createdAt }))],
    ['customers', 'id bigint, ext_id text, user_id bigint, name text, email text, phone text, type customer_type, loyalty_points integer, notes text, created_at timestamptz',
      s.customers.map((c, i) => ({ id: i + 1, ext_id: c.id, user_id: ref(ids.users, c.userId), name: c.name, email: c.email, phone: c.phone,
        type: c.type, loyalty_points: c.loyaltyPoints, notes: c.notes, created_at: c.createdAt }))],
    ['customer_preferences', 'customer_id bigint, kind text, value text',
      s.customers.flatMap((c, i) => [
        ...(c.preferences?.dietary ?? []).map((value) => ({ customer_id: i + 1, kind: 'dietary', value })),
        ...(c.preferences?.allergies ?? []).map((value) => ({ customer_id: i + 1, kind: 'allergy', value })),
      ]), { identity: false }],
  ];
}

export function load(r, x) {
  const prefs = (id, kind) => r.customer_preferences.filter((p) => p.customer_id === id && p.kind === kind).map((p) => p.value);
  return {
    users: r.users.map((u) => ({ id: u.ext_id, name: u.name, email: u.email, role: u.role, phone: u.phone, active: u.active,
      mustChangePassword: u.must_change_password, tokenVersion: u.token_version, passwordHash: u.password_hash,
      hourlyWage: num(u.hourly_wage), createdAt: iso(u.created_at) })),
    customers: r.customers.map((c) => ({ id: c.ext_id, userId: ref(x.users, c.user_id), phone: c.phone, type: c.type,
      loyaltyPoints: c.loyalty_points, preferences: { dietary: prefs(c.id, 'dietary'), allergies: prefs(c.id, 'allergy') },
      notes: c.notes, createdAt: iso(c.created_at), name: c.name, email: c.email })),
  };
}
```

The remaining modules follow the same shape, with these columns (exact defs = the `CREATE TABLE` column list in `001_schema.sql`, types copied verbatim, enum columns typed by their enum name):

- `menu.js` — `menu_categories` (cat), `inventory_items`, `menu_items` (`category_id` via `ids.categories`, `dietary_tags text[]`, `allergens text[]`, `modifiers jsonb`), `recipe_lines` (`inventory_id` via `ids.inventory`, id = running counter). Load drops `created_at`; `recipe` = lines of that item ordered by id.
- `floor.js` — `tables` inserted with `reserved_for: null`; `reservations` (`time` → `hm` on load, `date` as text); then a fix-up statement `UPDATE tables t SET reserved_for = x.r FROM jsonb_to_recordset($1::jsonb) AS x(id bigint, r bigint) WHERE t.id = x.id`.
- `orders.js` — `orders` (`rates` → `tax_rate`/`service_charge_rate`), `order_items` (ids from a flattened `ids.orderItems`), `order_refunds` (`by` ↔ `by_user_id`), `bill_splits`/`split_parts` (`part_no` = index)/`split_part_items`; load rebuilds `items`, `rates`, `split` (or `null`), `refund` (or `null`) and calls `computeTotals(order)` from `../../lib/order-math.js`.
- `stock.js` — `stock_movements`, `purchase_orders` (`number` ↔ `PO-0001`), `purchase_order_lines` (incl. `supplier`).
- `alerts.js` — `notifications` (`to` ↔ `to_email`), `notification_reads` (`read_at` = notification `created_at`).
- `workforce.js` — `staff_applications`, `shifts` (`"start" time, "end" time`), `attendance_sessions`, `attendance_breaks`, `pay_adjustments`.
- `config.js` — `settings` (single row id 1, camelCase ↔ snake_case, `identity:false`), `sequences` rows `order_seq` and `id:<prefix>` per counter (`identity:false`).

```js
// index.js
import * as people from './people.js';
import * as menu from './menu.js';
import * as floor from './floor.js';
import * as orders from './orders.js';
import * as stock from './stock.js';
import * as alerts from './alerts.js';
import * as workforce from './workforce.js';
import * as config from './config.js';
import { idsOf, extsOf, insertRows, selectRows } from './sql.js';

const AREAS = [people, menu, floor, orders, stock, alerts, workforce, config];
const TABLES = ['users', 'customers', 'customer_preferences', 'menu_categories', 'inventory_items', 'menu_items', 'recipe_lines',
  'tables', 'reservations', 'orders', 'order_items', 'order_refunds', 'bill_splits', 'split_parts', 'split_part_items',
  'stock_movements', 'purchase_orders', 'purchase_order_lines', 'notifications', 'notification_reads', 'staff_applications',
  'shifts', 'attendance_sessions', 'attendance_breaks', 'pay_adjustments', 'loyalty_ledger', 'settings'];
const NO_ID = new Set(['customer_preferences', 'split_part_items', 'notification_reads', 'settings']);

export async function saveStore(db, s) {
  const ids = {
    users: idsOf(s.users), customers: idsOf(s.customers), categories: idsOf(s.categories), inventory: idsOf(s.inventory),
    menuItems: idsOf(s.menuItems), tables: idsOf(s.tables), reservations: idsOf(s.reservations), orders: idsOf(s.orders),
    orderItems: idsOf(s.orders.flatMap((o) => o.items)), purchaseOrders: idsOf(s.purchaseOrders),
    notifications: idsOf(s.notifications), shifts: idsOf(s.shifts), attendance: idsOf(s.attendance),
  };
  await db.query(`TRUNCATE ${TABLES.join(', ')} RESTART IDENTITY CASCADE`);
  await db.query("DELETE FROM sequences WHERE name <> 'state_version'");
  for (const area of AREAS) {
    for (const step of area.save(s, ids)) {
      if (typeof step === 'function') await step(db); else await insertRows(db, ...step);
    }
  }
}

export async function loadStore(db) {
  const rows = {};
  for (const t of TABLES) rows[t] = await selectRows(db, t, NO_ID.has(t) ? '1' : 'id');
  rows.sequences = await selectRows(db, 'sequences', 'name');
  const exts = {
    users: extsOf(rows.users), customers: extsOf(rows.customers), categories: extsOf(rows.menu_categories),
    inventory: extsOf(rows.inventory_items), menuItems: extsOf(rows.menu_items), tables: extsOf(rows.tables),
    reservations: extsOf(rows.reservations), orders: extsOf(rows.orders), orderItems: extsOf(rows.order_items),
    notifications: extsOf(rows.notifications), shifts: extsOf(rows.shifts), attendance: extsOf(rows.attendance_sessions),
  };
  return Object.assign({}, ...AREAS.map((a) => a.load(rows, exts)));
}
```

A `save` step may be a function `(db) => …` for fix-ups such as the `tables.reserved_for` update.
